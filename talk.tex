\section{Motivation \& Research Questions}

\begin{frame}{Motivation}
  \begin{quote}\itshape
    ``The visualization collections are inherently diverse in many perspectives: from different data sources, by different research groups, and in different manners.''
  \end{quote}
  \vskip0.85em
  \begin{quote}\itshape
    ``Existing systems for visualization images \dots\ are mainly based on extrinsic attributes including author names and publication years.''
  \end{quote}
  \vskip0.85em
  \begin{quote}\itshape
    ``More importantly, attribute-based systems neglect the intrinsic property, i.e., visual appearance of visualizations that is also perceived by humans.''
  \end{quote}
  \vfill
  {\color{lumenPrimary}\bfseries Although visual appearance is directly perceived by humans, it has often been overlooked as an analyzable property of visualization collections.\par}
  \vskip0.4em
  \begin{flushright}
    \scriptsize Y.~Ye, R.~Huang, W.~Zeng. \emph{VISAtlas}. IEEE TVCG 30(7):3224--3240, 2024.
  \end{flushright}
\end{frame}

\begin{frame}{Research Questions}
  \vfill
  \begin{quote}\itshape
    ``Which visualization collection shall be used?\\
    Are there any similar visualizations in the collection?\\
    What visual features are prominent in a visualization?\\
    What visual features are frequently composited in visualizations?''
  \end{quote}
  \vfill
  {\color{lumenPrimary}\bfseries VISAtlas represents the visual appearance of visualizations as interpretable embeddings, enabling users to explore, compare, and query visualization collections based on visual features rather than only extrinsic metadata.\par}
  \vskip0.4em
  \begin{flushright}
    \scriptsize Y.~Ye, R.~Huang, W.~Zeng. \emph{VISAtlas}. IEEE TVCG 30(7):3224--3240, 2024.
  \end{flushright}
\end{frame}

\begin{frame}{Does Real-World Data Actually Help?}
  \begin{columns}[T]
  \begin{column}{0.55\textwidth}
  \centering
  \includegraphics[width=0.95\linewidth]{figures/eval_model_perf.png}
  \end{column}
  \begin{column}{0.45\textwidth}
  \small
  \begin{itemize}
    \item Overall F1: \textbf{82.3\%} (synthetic only) $\rightarrow$ \textbf{94.4\%} (synthetic + real), same CNN
    \item Biggest gains: \textbf{Map} and \textbf{Diagram} --- real maps have texture and noise a clean D3 render never has
    \item CNNs clearly beat classic ML (SVM, Random Forest)
    \item Triplet loss adds another $\sim$2 points on top of plain classification
  \end{itemize}
  \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Case Studies 2 \& 3 --- Composites and Retrieval}
  \textbf{Study 2: finding composite visualizations}
  \begin{itemize}\small
    \item Filter for points \emph{without} a dominant embedding dimension (e.g.\ Matrix $\approx$ Table $\approx 0.4$--$0.5$)
    \item Found: table visualizations with embedded heatmap cells --- genuinely composite designs a keyword search would never surface
  \end{itemize}
  \vspace{2mm}
  \textbf{Study 3: querying with real vs.\ hand-drawn sketches}
  \begin{itemize}\small
    \item Robust to different colormaps and layouts for \emph{real} images
    \item A rough hand-drawn pie chart came back misclassified as a spiral line graph --- wiggly strokes fall outside the training distribution
  \end{itemize}
\end{frame}

\begin{frame}{Our Critical Take}
  \begin{block}{What this suggests}
  The embedding may respond more to \textbf{low-level visual salience} (colour saturation, contrast, edge density) than to the actual \emph{structural semantics} of a chart type.
  \end{block}
  \vspace{2mm}
  \small
  \begin{itemize}
    \item The paper frames its embeddings as ``human-interpretable'' --- our minimal-pair test suggests that interpretation can be steered by pure rendering style, independent of content
    \item This goes a step further than the paper's own admission that the model doesn't fully capture ``colormaps and layout arrangements''
    \item \textbf{Discussion question:} if a CNN classifier can be flipped by style alone, how much should we trust embedding \emph{distance} as a proxy for design \emph{similarity}?
  \end{itemize}
\end{frame}

\section{Impact \& Conclusion}
% =====================================================

\begin{frame}{From One Paper to a Research Programme}
  \centering
  \includegraphics[width=0.85\textwidth]{figures/citation_provenance.pdf}\\[1mm]
  \footnotesize\color{lumenSlate} The core idea --- contextual, anchor-based projection of neural embeddings --- generalises from single-modality chart images (2022) to cross-modal LLM evaluation (2026), and to entirely new domains such as computational art criticism.
\end{frame}

\begin{frame}{Take-Home Messages}
  \begin{enumerate}
    \setlength\itemsep{4mm}
    \item VISAtlas shows that \textbf{visual appearance}, not just metadata, is a legitimate and useful axis for exploring large chart collections
    \item Its main trick --- a \textbf{fixed, human-labelled anchor layout} (RadViz + CNN) --- keeps a high-dimensional embedding interpretable, at the cost of distorting true distances
    \item Its own authors, and our experiments, agree: the taxonomy and CNN backbone are still \textbf{too coarse} to fully capture design similarity
    \item The idea proved durable: it is the direct ancestor of a still-active line of multimodal-embedding-visualization research
  \end{enumerate}
\end{frame}
